Producer sheet -> Extensions -> Apps Script -> delete everything -> paste -> Save. Deploy -> Manage deployments -> Edit -> New version -> Deploy.

/**
 * SetPulse - Producer signup + MC deck + OBS
 *
 * Bind this script to the Producer Signup spreadsheet.
 *
 * Tabs
 *   SIGNUPS  raw public form / list / QR / IG entries
 *   MC_DECK  what the MC edits (order, status, walk-ups)
 *   LINEUP   producer / OBS board (kept in sync with MC_DECK)
 *
 * Deploy
 *   1. Extensions → Apps Script → paste this file → Save
 *   2. Deploy → New deployment → Web app
 *      Execute as: Me
 *      Who has access: Anyone
 *   3. Paste the web app URL into SHARE!B7 and into the public list / MC HTML pages
 *
 * Public form
 *   Link the Google Form responses to this spreadsheet.
 *   If Google names the tab "Form Responses 1", either rename it SIGNUPS
 *   or set TAB_SIGNUPS below to that name.
 *
 * POST body (JSON)
 *   { action:"queue", name, note, show, source, chat_name, order }
 *   { action:"insert"|"move", name, order }   put name at that spot; everyone at/after slides down
 *   { action:"now"|"on_stage", name }
 *   { action:"next", name }
 *   { action:"done", name }
 *   { action:"sync" }          copy SIGNUPS → MC_DECK → LINEUP
 *
 * GET
 *   ?format=obs            now / next / on_deck
 *   ?format=full           lineup + obs
 *   ?page=signup           public HTML form
 *   ?page=mc               MC phone deck
 */

const TAB_SIGNUPS = 'SIGNUPS';
const TAB_MC = 'MC_DECK';
const TAB_LINEUP = 'LINEUP';
const DEFAULT_SHOW = 'Comedy Simulator';
const SLOT_MAX = 20;
const MC_PIN = '4177';
const PROTECTED_ACTIONS = ['insert', 'move', 'now', 'on_stage', 'now_performing', 'next', 'done', 'clear', 'cut', 'no_show', 'sync', 'open_signups', 'close_signups', 'begin_show', 'cue_next'];
const SIGNUPS_OPEN_PROP = 'signups_open';

function signupsOpen_() {
  return PropertiesService.getScriptProperties().getProperty(SIGNUPS_OPEN_PROP) === 'true';
}

function setSignupsOpen_(open) {
  PropertiesService.getScriptProperties().setProperty(SIGNUPS_OPEN_PROP, open ? 'true' : 'false');
}

const SIGNUP_HEADERS = ['Timestamp', 'Stage Name', 'Chat Name', 'Note', 'Source', 'Show', 'Channel', 'Synced', 'Signup ID'];
const MC_HEADERS = ['Order', 'Stage Name', 'Chat Name', 'Note', 'Show', 'Status', 'Added', 'On stage', 'Done', 'Source', 'MC note'];
const LINEUP_HEADERS = ['order', 'name', 'note', 'show', 'status', 'added_at', 'on_stage_at', 'done_at', 'source'];

function onOpen() {
  SpreadsheetApp.getUi()
    .createMenu('SetPulse')
    .addItem('Sync signups → MC → LINEUP', 'syncAll')
    .addItem('Renumber list 1...n', 'renumberList')
    .addItem('Ensure tabs', 'ensureTabs')
    .addToUi();
}

function onFormSubmit(e) {
  // Google Forms usually writes the response before this runs.
  syncSignupsToMc_();
  mirrorMcToLineup_();
}

function onEdit(e) {
  if (!e || !e.range) return;
  if (e.range.getSheet().getName() !== TAB_MC) return;
  const col = e.range.getColumn();
  const row = e.range.getRow();
  if (row < 3) return;
  const sh = e.range.getSheet();
  const stage = String(sh.getRange(row, 2).getValue() || '').trim();
  if (col === 6) stampStatus_(sh, row, String(e.value || ''));
  // Order cell changed on a named row → move that comic into the new spot; others slide.
  if (col === 1 && stage) {
    const dest = parseInt(e.value, 10);
    if (dest > 0) placeAt_(stage, dest);
  }
  // New name typed onto a row that already has an Order → insert at that spot.
  if (col === 2 && stage) {
    const dest = parseInt(sh.getRange(row, 1).getValue(), 10);
    if (dest > 0) placeAt_(stage, dest);
    else addMcRow_(stage, String(sh.getRange(row, 4).getValue() || ''), DEFAULT_SHOW, 'mc', '');
  }
  mirrorMcToLineup_();
}

function ensureTabs() {
  const ss = SpreadsheetApp.getActive();
  sheetWithHeaders_(ss, TAB_SIGNUPS, SIGNUP_HEADERS);
  sheetWithHeaders_(ss, TAB_MC, MC_HEADERS);
  sheetWithHeaders_(ss, TAB_LINEUP, LINEUP_HEADERS);
}

function sheetWithHeaders_(ss, tab, headers) {
  let sh = ss.getSheetByName(tab);
  if (!sh) sh = ss.insertSheet(tab);
  const existing = sh.getRange(2, 1, 1, headers.length).getValues()[0];
    // Row 1 is a banner on the xlsx; headers live on row 2.
  const row1 = sh.getRange(1, 1).getValue();
  const headerRow = (String(row1).toLowerCase().indexOf('order') === 0 || String(row1).toLowerCase().indexOf('timestamp') === 0) ? 1 : 2;
  const have = sh.getRange(headerRow, 1, 1, headers.length).getValues()[0].join('|').toLowerCase();
  if (!have || have.split('|').filter(Boolean).length < 3) {
    sh.getRange(headerRow, 1, 1, headers.length).setValues([headers]);
    sh.setFrozenRows(headerRow);
  }
  return { sh: sh, headerRow: headerRow };
}

function headerRowOf_(sh, fallback) {
  const a1 = String(sh.getRange(1, 1).getValue() || '').toLowerCase();
  if (a1 === 'order' || a1 === 'timestamp' || a1 === 'order'.toLowerCase()) return 1;
  return fallback || 2;
}

function mcSheet_() {
  return sheetWithHeaders_(SpreadsheetApp.getActive(), TAB_MC, MC_HEADERS);
}

function lineupSheet_() {
  return sheetWithHeaders_(SpreadsheetApp.getActive(), TAB_LINEUP, LINEUP_HEADERS);
}

function signupsSheet_() {
  const ss = SpreadsheetApp.getActive();
  let sh = ss.getSheetByName(TAB_SIGNUPS);
  if (!sh) sh = ss.getSheetByName('Form Responses 1') || ss.insertSheet(TAB_SIGNUPS);
  return sheetWithHeaders_(ss, sh.getName(), SIGNUP_HEADERS);
}

function readNamedRows_(sh, headerRow, mapFn) {
  const last = sh.getLastRow();
  if (last <= headerRow) return [];
  const width = sh.getLastColumn();
  const values = sh.getRange(headerRow + 1, 1, last - headerRow, width).getValues();
  return values.map((r, i) => mapFn(r, headerRow + 1 + i)).filter(Boolean);
}

function mcRows_() {
  const { sh, headerRow } = mcSheet_();
  return readNamedRows_(sh, headerRow, (r, row) => {
    const name = String(r[1] || '').trim();
    if (!name) return null;
    return {
      row: row,
      order: r[0],
      name: name,
      chat_name: String(r[2] || ''),
      note: String(r[3] || ''),
      show: String(r[4] || ''),
      status: String(r[5] || 'waiting'),
      added_at: r[6],
      on_stage_at: r[7],
      done_at: r[8],
      source: r[9],
      mc_note: r[10]
    };
  });
}

function signupRows_() {
  const { sh, headerRow } = signupsSheet_();
  // Support both our schema and a raw Google Form schema
  // Our: Timestamp | Stage Name | Chat Name | Note | Source | Show | Channel | Synced | Signup ID
  // Form: Timestamp | Stage name | Chat name | Note
  return readNamedRows_(sh, headerRow, (r, row) => {
    const name = String(r[1] || '').trim();
    if (!name) return null;
    return {
      row: row,
      timestamp: r[0],
      name: name,
      chat_name: String(r[2] || ''),
      note: String(r[3] || ''),
      source: String(r[4] || 'form'),
      show: String(r[5] || DEFAULT_SHOW),
      channel: String(r[6] || 'form'),
      synced: String(r[7] || 'no').toLowerCase()
    };
  });
}

function isLiveStatus_(status) {
  const s = String(status || '').toLowerCase();
  return s !== 'done' && s !== 'cut' && s !== 'no_show';
}

function takenSlots_() {
  const map = {};
  liveNamed_().forEach(r => {
    if (!isLiveStatus_(r.status)) return;
    const o = parseInt(r.order, 10);
    if (o >= 1 && o <= SLOT_MAX) map[o] = r.name;
  });
  return map;
}

function slotsPayload_() {
  const taken = takenSlots_();
  const slots = [];
  for (let i = 1; i <= SLOT_MAX; i++) {
    slots.push({ order: i, open: !taken[i], name: taken[i] || '' });
  }
  return slots;
}

function firstOpenSlot_() {
  const taken = takenSlots_();
  for (let i = 1; i <= SLOT_MAX; i++) if (!taken[i]) return i;
  return 0;
}

function occupySlot_(name, note, show, source, chatName, atOrder) {
  const dest = parseInt(atOrder, 10);
  if (!(dest >= 1 && dest <= SLOT_MAX)) {
    return { ok: false, error: 'pick a slot 1-20', slots: slotsPayload_() };
  }
  const taken = takenSlots_();
  const existing = liveNamed_().find(r => r.name.toLowerCase() === String(name).toLowerCase() && isLiveStatus_(r.status));
  if (existing && Number(existing.order) === dest) {
    return { ok: true, already: true, order: dest, slots: slotsPayload_() };
  }
  if (taken[dest] && (!existing || taken[dest].toLowerCase() !== String(name).toLowerCase())) {
    return { ok: false, error: 'slot ' + dest + ' is taken', slots: slotsPayload_() };
  }
  if (existing) {
    const { sh } = mcSheet_();
    sh.getRange(existing.row, 1).setValue(dest);
    return { ok: true, order: dest, slots: slotsPayload_() };
  }
  const { sh } = mcSheet_();
  sh.appendRow([dest, name, chatName || '', note || '', show || DEFAULT_SHOW, 'waiting', new Date(), '', '', source || 'list', '']);
  return { ok: true, order: dest, slots: slotsPayload_() };
}

function addMcRow_(name, note, show, source, chatName, atOrder) {
  const all = mcRows_();
  const live = all.find(r => r.name.toLowerCase() === name.toLowerCase() && isLiveStatus_(r.status));
  if (live) {
    if (atOrder) placeAt_(name, atOrder);
    return live;
  }
  const { sh } = mcSheet_();
  const now = new Date();
  const dest = parseInt(atOrder, 10);
  const order = dest > 0 ? Math.min(dest, SLOT_MAX) : (firstOpenSlot_() || all.length + 1);
  sh.appendRow([order, name, chatName || '', note || '', show || DEFAULT_SHOW, 'waiting', now, '', '', source || 'form', '']);
  if (dest > 0) placeAt_(name, dest);
  return { name: name, status: 'waiting', order: order };
}

function liveNamed_() {
  return mcRows_().slice().sort((a, b) => Number(a.order || 9999) - Number(b.order || 9999));
}

function placeAt_(name, dest) {
  const target = Math.max(1, parseInt(dest, 10) || 1);
  const { sh } = mcSheet_();
  const named = liveNamed_();
  const moving = named.find(r => r.name.toLowerCase() === String(name).toLowerCase());
  if (!moving) return;
  const without = named.filter(r => r.row !== moving.row);
  const idx = Math.min(Math.max(target - 1, 0), without.length);
  without.splice(idx, 0, moving);
  without.forEach((r, i) => sh.getRange(r.row, 1).setValue(i + 1));
}

function compactOrders_() {
  const { sh } = mcSheet_();
  liveNamed_().forEach((r, i) => {
    if (Number(r.order) !== i + 1) sh.getRange(r.row, 1).setValue(i + 1);
  });
}

function renumberList() {
  compactOrders_();
  mirrorMcToLineup_();
}

function stampStatus_(sh, row, status) {
  const now = new Date();
  const s = String(status || '').toLowerCase();
  if (s === 'on_stage') sh.getRange(row, 8).setValue(now);
  if (s === 'done' || s === 'cut' || s === 'no_show') sh.getRange(row, 9).setValue(now);
}

function writeMcStatus_(name, status) {
  const { sh } = mcSheet_();
  const all = mcRows_();
  const now = new Date();
  let hit = all.find(r => r.name.toLowerCase() === name.toLowerCase());
  if (!hit) {
    addMcRow_(name, '', DEFAULT_SHOW, 'mc', '');
    hit = mcRows_().find(r => r.name.toLowerCase() === name.toLowerCase());
  }
  if (!hit) return;
  sh.getRange(hit.row, 6).setValue(status);
  stampStatus_(sh, hit.row, status);
}

function syncSignupsToMc_() {
  const { sh, headerRow } = signupsSheet_();
  const rows = signupRows_();
  rows.forEach(r => {
    if (r.synced === 'yes') return;
    addMcRow_(r.name, r.note, r.show, r.source || 'form', r.chat_name);
    // Synced column is H (8) on our schema; skip if the form tab has fewer columns
    if (sh.getLastColumn() >= 8) sh.getRange(r.row, 8).setValue('yes');
  });
}

function mirrorMcToLineup_() {
  const { sh, headerRow } = lineupSheet_();
  const rows = mcRows_().sort((a, b) => Number(a.order || 0) - Number(b.order || 0));
  const last = sh.getLastRow();
  if (last > headerRow) sh.getRange(headerRow + 1, 1, last - headerRow, LINEUP_HEADERS.length).clearContent();
  if (!rows.length) return;
  const values = rows.map(r => [
    r.order, r.name, r.note, r.show || DEFAULT_SHOW, r.status || 'waiting',
    r.added_at, r.on_stage_at, r.done_at, r.source || ''
  ]);
  sh.getRange(headerRow + 1, 1, values.length, LINEUP_HEADERS.length).setValues(values);
}

function syncAll() {
  ensureTabs();
  syncSignupsToMc_();
  mirrorMcToLineup_();
}

function obsPayload_() {
  const all = mcRows_().sort((a, b) => Number(a.order || 0) - Number(b.order || 0));
  const waiting = all.filter(r => r.status === 'waiting' || r.status === 'next');
  const on = all.find(r => r.status === 'on_stage');
  const next = all.find(r => r.status === 'next') || waiting[0] || null;
  return {
    now: on ? on.name : '',
    next: next && (!on || next.name !== on.name) ? next.name : (waiting[1] ? waiting[1].name : ''),
    on_deck: waiting.slice(0, 8).map(r => r.name),
    show: (on && on.show) || (waiting[0] && waiting[0].show) || DEFAULT_SHOW
  };
}

function readPayload_() {
  return { obs: obsPayload_(), slots: slotsPayload_(), signups_open: signupsOpen_() };
}

function handleAction_(params) {
  const action = String(params.action || params.status || '').toLowerCase();
  const name = String(params.name || params.current_performer || '').trim();
  const note = String(params.note || '');
  const show = String(params.show || DEFAULT_SHOW);
  const source = String(params.source || 'list');
  const chat = String(params.chat_name || params.chat || '');
  const channel = String(params.channel || source);
  const atOrder = parseInt(params.order || params.at || params.spot || 0, 10);
  const pin = String(params.pin || '');

  if (PROTECTED_ACTIONS.indexOf(action) !== -1 && pin !== MC_PIN) {
    return { ok: false, error: 'wrong or missing PIN' };
  }

  if (action === 'open_signups') {
    setSignupsOpen_(true);
    return Object.assign({ ok: true }, readPayload_());
  }
  if (action === 'close_signups') {
    setSignupsOpen_(false);
    return Object.assign({ ok: true }, readPayload_());
  }

  if (action === 'begin_show') {
    const waiting = liveNamed_().filter(r => r.status === 'waiting' || r.status === 'next');
    if (waiting.length) {
      mcRows_().filter(r => r.status === 'on_stage').forEach(r => writeMcStatus_(r.name, 'done'));
      writeMcStatus_(waiting[0].name, 'on_stage');
      mirrorMcToLineup_();
    }
    return Object.assign({ ok: true }, readPayload_());
  }

  if (action === 'cue_next') {
    const waiting = liveNamed_().filter(r => r.status === 'waiting');
    if (waiting.length) {
      writeMcStatus_(waiting[0].name, 'next');
      mirrorMcToLineup_();
    }
    return Object.assign({ ok: true }, readPayload_());
  }

  if (action === 'sync') {
    syncAll();
    return Object.assign({ ok: true }, readPayload_());
  }

  if (action === 'queue' || action === 'list' || action === 'waiting' || action === 'signup' || action === 'claim') {
    if (!signupsOpen_()) return Object.assign({ ok: false, error: 'Signups are not open yet.' }, readPayload_());
    if (!name) return Object.assign({ ok: false, error: 'name required' }, readPayload_());
    const claimed = occupySlot_(name, note, show, source, chat, atOrder || firstOpenSlot_());
    if (!claimed.ok) return Object.assign({}, claimed, readPayload_());
    appendSignup_(name, chat, note, source, show, channel);
    mirrorMcToLineup_();
    return Object.assign({ ok: true, order: claimed.order }, readPayload_());
  } else if (action === 'insert' || action === 'move') {
    if (!name) return Object.assign({ ok: false, error: 'name required' }, readPayload_());
    addMcRow_(name, note, show, source || 'mc', chat, atOrder || 1);
    placeAt_(name, atOrder || 1);
    compactOrders_();
    mirrorMcToLineup_();
  } else if (action === 'now' || action === 'on_stage' || action === 'now_performing') {
    if (!name) return Object.assign({ ok: false, error: 'name required' }, readPayload_());
    mcRows_().filter(r => r.status === 'on_stage').forEach(r => writeMcStatus_(r.name, 'done'));
    addMcRow_(name, note, show, 'mc', chat);
    writeMcStatus_(name, 'on_stage');
    mirrorMcToLineup_();
  } else if (action === 'next') {
    if (name) writeMcStatus_(name, 'next');
    mirrorMcToLineup_();
  } else if (action === 'done' || action === 'clear') {
    if (name) writeMcStatus_(name, 'done');
    else mcRows_().filter(r => r.status === 'on_stage').forEach(r => writeMcStatus_(r.name, 'done'));
    mirrorMcToLineup_();
  } else if (action === 'cut' || action === 'no_show') {
    if (name) writeMcStatus_(name, action);
    mirrorMcToLineup_();
  }

  return Object.assign({ ok: true }, readPayload_());
}

function doGet(e) {
  const params = (e && e.parameter) || {};
  const page = params.page || '';
  const pin = params.pin || '';
  if (page === 'signup' || page === 'list') return htmlPage_('signup');
  if (page === 'mc') return htmlPage_('mc', pin);

  const callback = params.callback;
  let out;
  if (params.action) {
    out = handleAction_(params);
  } else {
    const fmt = params.format || 'obs';
    if (fmt === 'full') out = Object.assign({ lineup: liveNamed_() }, readPayload_());
    else if (fmt === 'slots') out = readPayload_();
    else out = obsPayload_();
  }

  if (callback) {
    const safeCallback = String(callback).replace(/[^A-Za-z0-9_$.]/g, '');
    return ContentService.createTextOutput(safeCallback + '(' + JSON.stringify(out) + ')')
      .setMimeType(ContentService.MimeType.JAVASCRIPT);
  }
  return ContentService.createTextOutput(JSON.stringify(out)).setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  let body = {};
  try { body = JSON.parse((e.postData && e.postData.contents) || '{}'); } catch (err) { body = {}; }
  // Also accept form-encoded posts from a simple HTML form
  if (e && e.parameter && e.parameter.name && !body.name) {
    body = e.parameter;
  }
  if (!body.action) body.action = 'queue';
  return json_(handleAction_(body));
}

function appendSignup_(name, chat, note, source, show, channel) {
  const { sh, headerRow } = signupsSheet_();
  const id = Utilities.formatDate(new Date(), Session.getScriptTimeZone() || 'Pacific/Honolulu', "yyyyMMdd-HHmmss") + '-' + name.replace(/\s+/g, '').slice(0, 8);
  sh.appendRow([new Date(), name, chat || '', note || '', source || 'form', show || DEFAULT_SHOW, channel || '', 'yes', id]);
}

function json_(obj) {
  return ContentService.createTextOutput(JSON.stringify(obj)).setMimeType(ContentService.MimeType.JSON);
}

function htmlPage_(kind, pin) {
  const title = kind === 'mc' ? 'SetPulse MC' : 'Get on the list';
  const show = DEFAULT_SHOW;
  const html = kind === 'mc' ? mcHtml_(show, pin) : signupHtml_(show);
  return HtmlService.createHtmlOutput(html)
    .setTitle(title)
    .setXFrameOptionsMode(HtmlService.XFrameOptionsMode.ALLOWALL)
    .addMetaTag('viewport', 'width=device-width, initial-scale=1, viewport-fit=cover');
}

      document.getElementById('name').value='';
      document.getElementById('chat').value='';
      document.getElementById('note').value='';
      pick=0;
    }else status.textContent=j.error||'That spot is gone.';
    if(j.slots)draw(j.slots); else load();
  }catch(e){status.textContent='Network miss. Try again.';}
};
load();
setInterval(load,8000);
</script></body></html>`;
}
